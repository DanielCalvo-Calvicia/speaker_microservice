import pytest
from fastapi import FastAPI
from fastapi.testclient import TestClient

from infrastructure.inbound.http.http_auth import ApiKeyAuthenticator
from infrastructure.inbound.http.http_envelope import failure_message, success
from infrastructure.inbound.http.http_error_mapper import map_error


def _client() -> TestClient:
    app = FastAPI()
    from fastapi import Depends

    @app.get("/secret", dependencies=[Depends(ApiKeyAuthenticator("s3cret"))])
    def secret() -> dict[str, bool]:
        return {"ok": True}

    return TestClient(app)


def test_correct_bearer_key_is_accepted():
    assert _client().get("/secret", headers={"Authorization": "Bearer s3cret"}).status_code == 200


def test_missing_key_is_rejected_with_401():
    response = _client().get("/secret")
    assert response.status_code == 401
    assert response.headers["www-authenticate"] == "Bearer"


def test_wrong_key_is_rejected_with_401():
    assert _client().get("/secret", headers={"Authorization": "Bearer nope"}).status_code == 401


def test_empty_expected_key_is_refused_at_construction():
    with pytest.raises(ValueError):
        ApiKeyAuthenticator("")


def test_every_error_maps_to_500():
    assert map_error(ValueError("x")) == 500


def test_envelopes_carry_the_documented_fields():
    ok = success("act", "done", 1)
    assert ok.status_code == 200
    err = failure_message("act", "boom", 500, None)
    assert err.status_code == 500
    assert b'"status":"error"' in err.body
